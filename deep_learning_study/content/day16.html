1. Dataset: 데이터 한 개를 꺼내는 규칙

Dataset은 두 메서드만 구현하면 됩니다: __len__(개수), __getitem__(i)(i번째 샘플).

import torch
from torch.utils.data import Dataset, DataLoader, random_split
import pandas as pd
from PIL import Image

class CSVDataset(Dataset):
    """표 형식 데이터: 마지막 열이 레이블"""
    def __init__(self, path, mean=None, std=None):
        df = pd.read_csv(path)
        self.X = torch.tensor(df.iloc[:, :-1].values, dtype=torch.float32)
        self.y = torch.tensor(df.iloc[:, -1].values, dtype=torch.long)
        self.mean = self.X.mean(0) if mean is None else mean     # 학습 세트 통계를 검증에 재사용
        self.std = self.X.std(0) + 1e-8 if std is None else std
    def __len__(self): return len(self.y)
    def __getitem__(self, i):
        return (self.X[i] - self.mean) / self.std, self.y[i]

class ImageFolderDataset(Dataset):
    """이미지 파일 목록 + 레이블. 이미지는 __getitem__에서 그때그때 읽는다 (메모리 절약)"""
    def __init__(self, paths, labels, transform=None):
        self.paths, self.labels, self.transform = paths, labels, transform
    def __len__(self): return len(self.paths)
    def __getitem__(self, i):
        img = Image.open(self.paths[i]).convert("RGB")
        if self.transform: img = self.transform(img)
        return img, self.labels[i]

폴더 구조가 root/클래스명/이미지.jpg라면 torchvision.datasets.ImageFolder(root, transform)가 이 일을 해 줍니다.

  • shuffle=True: 학습용만. 검증은 섞을 필요 없음.
  • num_workers: 데이터를 읽는 병렬 프로세스 수. GPU가 데이터를 기다리며 놀지 않게 합니다.
  • pin_memory=True: CPU→GPU 전송 가속.
  • drop_last=True: 마지막 자투리 배치 버림(BN이 배치 크기 1에서 에러 나는 것 방지).
  • 샘플마다 길이가 다르면(문장 등) collate_fn으로 패딩해서 묶습니다(Day 21).

import random, numpy as np
def seed_everything(seed=42):
    random.seed(seed); np.random.seed(seed); torch.manual_seed(seed); torch.cuda.manual_seed_all(seed)
    torch.backends.cudnn.deterministic = True     # 완전 재현이 필요할 때 (약간 느려짐)
    torch.backends.cudnn.benchmark = False
seed_everything(42)
💡 시드는 하나로는 부족하다 같은 설정도 시드에 따라 정확도가 ±0.5% 넘게 흔들릴 수 있습니다. "A가 B보다 0.3% 좋다"는 결론을 내리려면 시드 3개 이상의 평균과 표준편차를 비교하세요.

4. 학습·평가 함수 템플릿

from tqdm.auto import tqdm

def train_one_epoch(model, loader, loss_fn, opt, device, scheduler=None, clip=None):
    model.train()
    total_loss, correct, n = 0.0, 0, 0
    for xb, yb in tqdm(loader, leave=False):
        xb, yb = xb.to(device, non_blocking=True), yb.to(device, non_blocking=True)
        logits = model(xb)
        loss = loss_fn(logits, yb)
        opt.zero_grad(set_to_none=True)
        loss.backward()
        if clip: torch.nn.utils.clip_grad_norm_(model.parameters(), clip)   # 기울기 클리핑
        opt.step()
        if scheduler: scheduler.step()                                       # 스텝 단위 스케줄러
        total_loss += loss.item() * len(yb)
        correct += (logits.argmax(1) == yb).sum().item(); n += len(yb)
    return total_loss / n, correct / n

@torch.no_grad()                      # 데코레이터로 함수 전체에 no_grad
def evaluate(model, loader, loss_fn, device):
    model.eval()
    total_loss, correct, n = 0.0, 0, 0
    for xb, yb in loader:
        xb, yb = xb.to(device), yb.to(device)
        logits = model(xb)
        total_loss += loss_fn(logits, yb).item() * len(yb)
        correct += (logits.argmax(1) == yb).sum().item(); n += len(yb)
    return total_loss / n, correct / n

5. 체크포인트: 저장과 불러오기

def save_ckpt(path, model, opt, epoch, best_val):
    torch.save({"model": model.state_dict(), "opt": opt.state_dict(), "epoch": epoch, "best_val": best_val}, path)

def load_ckpt(path, model, opt=None, device="cpu"):
    ck = torch.load(path, map_location=device)
    model.load_state_dict(ck["model"])
    if opt: opt.load_state_dict(ck["opt"])       # 옵티마이저 상태(Adam의 m, v)까지 복원해야 이어서 학습 가능
    return ck["epoch"], ck["best_val"]
🔑 state_dict를 저장하라 torch.save(model)로 모델 객체 전체를 저장하면 클래스 정의와 파일 경로에 묶여 나중에 깨지기 쉽습니다. 파라미터 사전(state_dict)만 저장하고, 불러올 때는 모델을 새로 만든 뒤 load_state_dict하세요. Colab은 세션이 끊기면 파일이 사라지므로 Google Drive에 저장하세요.

7. 전체 조립: 표준 학습 스크립트

history = {"tr_loss": [], "tr_acc": [], "va_loss": [], "va_acc": [], "lr": []}
best_val, patience, bad = 0.0, 5, 0

for epoch in range(EPOCHS):
    tr_loss, tr_acc = train_one_epoch(model, train_dl, loss_fn, opt, device, scheduler=sched)
    va_loss, va_acc = evaluate(model, val_dl, loss_fn, device)
    for k, v in zip(history, [tr_loss, tr_acc, va_loss, va_acc, opt.param_groups[0]["lr"]]): history[k].append(v)
    print(f"[{epoch+1:02d}/{EPOCHS}] train {tr_loss:.4f}/{tr_acc:.3f}  val {va_loss:.4f}/{va_acc:.3f}")

    if va_acc > best_val:                            # 최고 모델 저장
        best_val, bad = va_acc, 0
        save_ckpt("best.pt", model, opt, epoch, best_val)
    else:
        bad += 1
        if bad >= patience:                          # 조기 종료
            print("Early stopping"); break

load_ckpt("best.pt", model)                          # 최고 시점으로 복원 후 테스트
