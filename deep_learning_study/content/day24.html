1. 데이터: 다음 글자 맞히기

텍스트 전체를 정수 ID 배열로 만든 뒤, 길이 block_size의 조각을 무작위로 뽑습니다. 타깃은 입력을 한 칸 민 것입니다.

text = open("input.txt").read()                  # 셰익스피어 약 100만 글자
chars = sorted(set(text)); V = len(chars)         # 어휘 = 등장하는 모든 글자 (약 65개)
stoi = {c: i for i, c in enumerate(chars)}; itos = {i: c for c, i in stoi.items()}
encode = lambda s: [stoi[c] for c in s]
decode = lambda ids: "".join(itos[i] for i in ids)

data = torch.tensor(encode(text), dtype=torch.long)
n = int(0.9 * len(data)); train_data, val_data = data[:n], data[n:]

def get_batch(split, block_size=256, batch_size=64):
    d = train_data if split == "train" else val_data
    ix = torch.randint(len(d) - block_size - 1, (batch_size,))
    x = torch.stack([d[i:i + block_size] for i in ix])            # (B, T)
    y = torch.stack([d[i + 1:i + block_size + 1] for i in ix])    # (B, T) 한 칸 민 정답
    return x.to(device), y.to(device)
🔑 한 샘플 = T개의 학습 문제 입력 "hello"(T=5)에서 모델은 동시에 다섯 문제를 풉니다: "h"→"e", "he"→"l", "hel"→"l", "hell"→"o", "hello"→다음 글자. Causal 마스크 덕분에 각 위치는 자기까지만 보므로 부정행위가 없습니다.

2. 모델 전체 코드

import math, torch, torch.nn as nn, torch.nn.functional as F
from dataclasses import dataclass

@dataclass
class GPTConfig:
    vocab_size: int = 65
    block_size: int = 256     # 최대 문맥 길이
    n_layer: int = 6
    n_head: int = 6
    d_model: int = 384
    dropout: float = 0.2

class CausalSelfAttention(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.n_head = cfg.n_head
        self.qkv = nn.Linear(cfg.d_model, 3 * cfg.d_model)      # Q, K, V를 한 번에
        self.proj = nn.Linear(cfg.d_model, cfg.d_model)          # W_O
        self.drop = nn.Dropout(cfg.dropout); self.p = cfg.dropout
    def forward(self, x):
        B, T, D = x.shape
        q, k, v = self.qkv(x).split(D, dim=2)
        q, k, v = [t.view(B, T, self.n_head, D // self.n_head).transpose(1, 2) for t in (q, k, v)]   # (B, h, T, dk)
        # 직접 쓰면: att = softmax((q @ k.transpose(-2,-1)) / sqrt(dk) + causal) @ v
        y = F.scaled_dot_product_attention(q, k, v, is_causal=True, dropout_p=self.p if self.training else 0)  # FlashAttention 자동 사용
        y = y.transpose(1, 2).contiguous().view(B, T, D)        # 헤드 합치기
        return self.drop(self.proj(y))

class MLP(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.net = nn.Sequential(nn.Linear(cfg.d_model, 4 * cfg.d_model), nn.GELU(),
                                 nn.Linear(4 * cfg.d_model, cfg.d_model), nn.Dropout(cfg.dropout))
    def forward(self, x): return self.net(x)

class Block(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.ln1, self.ln2 = nn.LayerNorm(cfg.d_model), nn.LayerNorm(cfg.d_model)
        self.attn, self.mlp = CausalSelfAttention(cfg), MLP(cfg)
    def forward(self, x):
        x = x + self.attn(self.ln1(x))      # 소통
        x = x + self.mlp(self.ln2(x))       # 생각
        return x

class GPT(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.cfg = cfg
        self.tok_emb = nn.Embedding(cfg.vocab_size, cfg.d_model)
        self.pos_emb = nn.Embedding(cfg.block_size, cfg.d_model)     # 학습형 위치 임베딩
        self.drop = nn.Dropout(cfg.dropout)
        self.blocks = nn.Sequential(*[Block(cfg) for _ in range(cfg.n_layer)])
        self.ln_f = nn.LayerNorm(cfg.d_model)
        self.head = nn.Linear(cfg.d_model, cfg.vocab_size, bias=False)
        self.head.weight = self.tok_emb.weight                       # 가중치 공유 (weight tying)
        self.apply(self._init)
    def _init(self, m):
        if isinstance(m, (nn.Linear, nn.Embedding)): nn.init.normal_(m.weight, 0.0, 0.02)
        if isinstance(m, nn.Linear) and m.bias is not None: nn.init.zeros_(m.bias)

    def forward(self, idx, targets=None):
        B, T = idx.shape
        pos = torch.arange(T, device=idx.device)
        x = self.drop(self.tok_emb(idx) + self.pos_emb(pos))      # (B, T, D)
        x = self.ln_f(self.blocks(x))
        logits = self.head(x)                                      # (B, T, V)
        loss = None if targets is None else F.cross_entropy(logits.view(-1, logits.size(-1)), targets.view(-1))
        return logits, loss

model = GPT(GPTConfig(vocab_size=V)).to(device)
print(f"{sum(p.numel() for p in model.parameters())/1e6:.2f}M 파라미터")   # 약 10.7M

3. 학습 루프

opt = torch.optim.AdamW(model.parameters(), lr=3e-4, betas=(0.9, 0.95), weight_decay=0.1)
max_iters, warmup = 5000, 200
def lr_at(it):                                          # 워밍업 + 코사인 감쇠 (Day 28)
    if it < warmup: return 3e-4 * (it + 1) / warmup
    r = (it - warmup) / (max_iters - warmup); return 3e-5 + 0.5 * (3e-4 - 3e-5) * (1 + math.cos(math.pi * r))

@torch.no_grad()
def estimate_loss(iters=50):
    model.eval(); out = {}
    for split in ("train", "val"):
        out[split] = torch.stack([model(*get_batch(split))[1] for _ in range(iters)]).mean().item()
    model.train(); return out

for it in range(max_iters):
    for g in opt.param_groups: g["lr"] = lr_at(it)
    x, y = get_batch("train")
    with torch.autocast(device_type="cuda", dtype=torch.bfloat16):     # 혼합 정밀도
        _, loss = model(x, y)
    opt.zero_grad(set_to_none=True)
    loss.backward()
    torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)            # 기울기 클리핑
    opt.step()
    if it % 500 == 0:
        l = estimate_loss(); print(f"iter {it}: train {l['train']:.3f}  val {l['val']:.3f}")

4. 생성 전략

@torch.no_grad()
def generate(model, idx, max_new_tokens, temperature=1.0, top_k=None, top_p=None):
    model.eval()
    for _ in range(max_new_tokens):
        idx_cond = idx[:, -model.cfg.block_size:]                 # 문맥 창을 넘으면 잘라냄
        logits, _ = model(idx_cond)
        logits = logits[:, -1, :] / temperature                    # 마지막 위치만 사용
        if top_k is not None:                                      # 상위 k개만 남기기
            v, _ = torch.topk(logits, top_k); logits[logits < v[:, [-1]]] = -float("inf")
        probs = F.softmax(logits, dim=-1)
        if top_p is not None:                                      # 누적 확률 p까지만 (nucleus)
            sp, si = torch.sort(probs, descending=True); cum = sp.cumsum(-1)
            sp[cum - sp > top_p] = 0; probs = torch.zeros_like(probs).scatter(-1, si, sp); probs /= probs.sum(-1, keepdim=True)
        nxt = torch.multinomial(probs, 1)                          # 샘플링
        idx = torch.cat([idx, nxt], dim=1)
    return idx

start = torch.tensor([encode("ROMEO:")], device=device)
print(decode(generate(model, start, 500, temperature=0.8, top_k=40)[0].tolist()))
전략방법특징
탐욕(Greedy)항상 argmax결정적, 반복 루프에 빠지기 쉬움
온도 샘플링로짓 / τ 후 샘플링τ↓ 보수적, τ↑ 다양하지만 엉뚱함
Top-k상위 k개 중에서만 샘플링꼬리의 이상한 토큰 제거
Top-p (nucleus)누적 확률 p가 될 때까지의 토큰만분포 모양에 따라 후보 수가 적응적. 실무 표준

5. 실제 LLM으로 가는 디테일

  • KV 캐시: 생성할 때 매 스텝 전체 문맥을 다시 계산하는 것은 낭비입니다. 이전 토큰들의 K, V를 저장해 두고 새 토큰의 Q만 계산합니다. 생성 속도를 수십 배 높이지만 메모리를 많이 씁니다(긴 문맥이 비싼 또 다른 이유).
  • 가중치 공유: 입력 임베딩과 출력층이 같은 행렬. 파라미터 절약 + 성능 향상.
  • 초기화: 잔차 경로로 들어가는 출력 투영층을 $1/\sqrt{2N}$로 작게 초기화(GPT-2)하면 깊은 모델이 안정적입니다.
  • BPE 토큰화: 문자 대신 서브워드를 쓰면 같은 문맥 길이에 훨씬 많은 내용이 들어갑니다(Day 21).
  • 스케일: 같은 코드에서 n_layer=12, n_head=12, d_model=768, 문맥 1024, GPT-2 토크나이저, 웹 텍스트 수십억 토큰 → GPT-2 small(124M).
