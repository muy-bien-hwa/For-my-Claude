• 층, 손실, 옵티마이저, 데이터 공급기, 학습기를 갖춘 작은 딥러닝 프레임워크를 설계하고 구현한다.
  • 모든 부품을 기울기 검증과 "작은 배치 외우기" 시험으로 검증한다.
  • 같은 프레임워크로 회귀, 이진 분류, 다중 분류 세 과제를 풀고 2주차의 고전 모델과 비교한다.
  • 모델의 가중치를 저장하고 다시 불러온다.

프로젝트 개요

3주차에서 퍼셉트론부터 시작해 순전파, 역전파, 자동 미분, 학습 루프를 차례로 만들었다. 이번 주간 프로젝트에서는 이것들을 하나의 재사용 가능한 패키지 minidl로 정리한다. 설계는 PyTorch의 이름과 구조를 따르므로, 5주차에 PyTorch로 넘어가면 같은 개념을 그대로 알아볼 수 있다.

1단계: 설계

부품 사이의 약속(인터페이스)을 먼저 정한다. 각 부품은 이 약속만 지키면 다른 부품의 내부를 몰라도 된다.

부품메서드PyTorch 대응

손실forward(Z, y) → 스칼라, backward() → $\partial L/\partial Z$nn.CrossEntropyLoss 등

DataLoader반복하면 (X_batch, y_batch)를 냄torch.utils.data.DataLoader

train()과 eval()은 지금은 하는 일이 없지만, 26일차의 드롭아웃과 25일차의 배치 정규화처럼 학습과 평가에서 다르게 동작하는 층을 위해 미리 둔다.

2단계: Module과 층

import numpy as np
import matplotlib.pyplot as plt
import io

class Module:
    training = True
    def forward(self, X): raise NotImplementedError
    def backward(self, G): raise NotImplementedError
    def params(self): return []                      # [(파라미터, 기울기), ...]
    def train(self): self.training = True; return self
    def eval(self): self.training = False; return self
    def __call__(self, X): return self.forward(X)    # model(X)로 호출

class Linear(Module):
    def __init__(self, d_in, d_out, rng=None, init="he"):
        rng = rng or np.random.default_rng(0)
        std = np.sqrt(2 / d_in) if init == "he" else np.sqrt(1 / d_in)
        self.W = rng.normal(0, std, (d_in, d_out)); self.b = np.zeros(d_out)
        self.dW = np.zeros_like(self.W); self.db = np.zeros_like(self.b)
    def forward(self, X):
        self.X = X; return X @ self.W + self.b
    def backward(self, G):
        self.dW[...] = self.X.T @ G; self.db[...] = G.sum(0)
        return G @ self.W.T
    def params(self): return [(self.W, self.dW), (self.b, self.db)]

class ReLU(Module):
    def forward(self, X):
        self.mask = X > 0; return X * self.mask
    def backward(self, G): return G * self.mask

class Tanh(Module):
    def forward(self, X):
        self.out = np.tanh(X); return self.out
    def backward(self, G): return G * (1 - self.out ** 2)

class Sigmoid(Module):
    def forward(self, X):
        self.out = 0.5 * (1 + np.tanh(0.5 * X)); return self.out
    def backward(self, G): return G * self.out * (1 - self.out)

class Sequential(Module):
    def __init__(self, *layers): self.layers = list(layers)
    def forward(self, X):
        for l in self.layers: X = l.forward(X)
        return X
    def backward(self, G):
        for l in reversed(self.layers): G = l.backward(G)
        return G
    def params(self): return [pg for l in self.layers for pg in l.params()]
    def train(self):
        for l in self.layers: l.train()
        return self
    def eval(self):
        for l in self.layers: l.eval()
        return self
    def state_dict(self):
        return {f"p{i}": p.copy() for i, (p, _) in enumerate(self.params())}
    def load_state_dict(self, sd):
        for i, (p, _) in enumerate(self.params()): p[...] = sd[f"p{i}"]
print("층 준비 완료")

3단계: 손실 함수

세 손실 모두 마지막 층의 출력(로짓 또는 예측값) $Z$를 받는다. 확률로 바꾸는 과정을 손실 안에 넣어야 로그-합-지수(11일차)와 로짓 형태의 BCE(10일차)로 안정하게 계산할 수 있다.

class SoftmaxCrossEntropy:                           # 다중 분류, y: 정수 (N,)
    def forward(self, Z, y):
        Z = Z - Z.max(1, keepdims=True)
        self.P = np.exp(Z) / np.exp(Z).sum(1, keepdims=True)
        self.y = y
        return -np.log(self.P[np.arange(len(y)), y]).mean()
    def backward(self):
        G = self.P.copy(); G[np.arange(len(self.y)), self.y] -= 1
        return G / len(self.y)

# 이진 분류, Z: (N, 1), y: 0/1 (N,)
class BCEWithLogits:
    def forward(self, Z, y):
        self.z = Z[:, 0]; self.y = y.astype(float)
        # ln(1 + e^z) − yz
        return np.mean(np.logaddexp(0, self.z) - self.y * self.z)
    def backward(self):
        p = 0.5 * (1 + np.tanh(0.5 * self.z))
        return ((p - self.y) / len(self.y))[:, None]

class MSE:                                           # 회귀, Z: (N, 1), y: (N,)
    def forward(self, Z, y):
        self.d = Z[:, 0] - y; return np.mean(self.d ** 2)
    def backward(self): return (2 * self.d / len(self.d))[:, None]
print("손실 준비 완료")

BCE와 MSE에서 Z[:, 0]으로 $(N, 1)$을 $(N,)$로 바꾼 뒤 y와 계산하고, 역전파에서는 다시 [:, None]으로 $(N, 1)$로 되돌린다. 17일차의 $(N,)$ 대 $(N, 1)$ 함정을 피하기 위해서다.

4단계: 옵티마이저와 DataLoader

옵티마이저는 (파라미터, 기울기) 쌍의 목록을 받아 기울기를 비우고(zero_grad) 갱신한다(step). 가중치 감쇠(12일차의 L2 규제)는 갱신식에 $\lambda\mathbf w$를 더하는 것으로 구현한다. 편의상 편향에도 적용한다.

class SGD:
    def __init__(self, params, lr=0.1, weight_decay=0.0):
        self.params, self.lr, self.wd = params, lr, weight_decay
    def zero_grad(self):
        for _, g in self.params: g[...] = 0
    def step(self):
        for p, g in self.params:
            p -= self.lr * (g + self.wd * p)        # TODO 확인: 제자리 갱신 + 감쇠

class DataLoader:
    def __init__(self, X, y, batch_size=32, shuffle=True, seed=0):
        self.X, self.y, self.B, self.shuffle = X, y, batch_size, shuffle
        self.rng = np.random.default_rng(seed)
    def __iter__(self):
        idx = self.rng.permutation(len(self.X)) if self.shuffle else np.arange(len(self.X))
        for s in range(0, len(idx), self.B):
            i = idx[s:s + self.B]; yield self.X[i], self.y[i]
    def __len__(self): return -(-len(self.X) // self.B)    # 올림 나눗셈
print("옵티마이저와 DataLoader 준비 완료")

5단계: 학습기

def evaluate(model, loss_fn, X, y, metric):
    model.eval(); Z = model(X); L = loss_fn.forward(Z, y); model.train()
    return L, metric(Z, y)

def fit(model, loss_fn, opt, loader, val=None, epochs=10, metric=None):
    hist = {"train_loss": [], "val_loss": [], "val_metric": []}
    for ep in range(epochs):
        model.train(); total, n = 0.0, 0
        for Xb, yb in loader:
            L = loss_fn.forward(model(Xb), yb)
            opt.zero_grad()
            model.backward(loss_fn.backward())
            opt.step()
            total += L * len(Xb); n += len(Xb)
        hist["train_loss"].append(total / n)          # 에폭 평균 학습 손실
        if val is not None:
            vl, vm = evaluate(model, loss_fn, *val, metric)
            hist["val_loss"].append(vl); hist["val_metric"].append(vm)
    return hist

def grad_check(model, loss_fn, X, y, h=1e-5):
    loss_fn.forward(model(X), y); model.backward(loss_fn.backward()); worst = 0.0
    for p, g in model.params():
        g_bp = g.copy()
        for idx in np.ndindex(p.shape):
            old = p[idx]
            p[idx] = old + h; Lp = loss_fn.forward(model(X), y)
            p[idx] = old - h; Lm = loss_fn.forward(model(X), y)
            p[idx] = old
            num = (Lp - Lm) / (2 * h)
            worst = max(worst,
                        abs(num - g_bp[idx]) / max(abs(num), abs(g_bp[idx]), 1e-12))
    return worst
print("학습기 준비 완료")

8단계: 저장과 불러오기

학습한 가중치를 파일로 저장해 두면 다시 학습하지 않고 쓸 수 있다. state_dict로 파라미터를 사전으로 모으고 NumPy의 savez로 저장한다. 여기서는 메모리 안의 파일(BytesIO)에 저장한다.

best = results["MLP 16 tanh + 감쇠"]

loaded = dict(np.load(buf))

                   Linear(16, 1, np.random.default_rng(99)))
print("불러오기 전 예측 일치:", np.allclose(fresh(Zw[te]), best(Zw[te])))

PyTorch에서는 torch.save(model.state_dict(), "model.pt")와 model.load_state_dict(torch.load("model.pt"))로 같은 일을 한다. 모델 구조(코드)와 가중치(파일)를 따로 저장하는 방식이다.

참고 보고서

minidl은 Module(forward/backward/params), 손실(forward/backward), SGD(zero_grad/step), DataLoader, fit으로 이루어지며, 부품은 "(파라미터, 기울기) 쌍"과 "상류 기울기 G"만 주고받는다. 세 손실 모두 신경망 전체의 기울기 검증에서 상대 오차 $10^{-8}$ 이하였고, 표본 16개를 200스텝 만에 손실 0.006까지 외웠다. 손글씨는 테스트 97.3%로 소프트맥스 회귀(97.0%)와 비슷했고, 유방암은 테스트 115개를 모두 맞혀 로지스틱 회귀(99.1%)와 비슷했으나 테스트 BCE는 더 낮았다. 와인 회귀에서는 은닉층 16개와 가중치 감쇠로 테스트 $R^2$가 선형 모델의 0.64에서 0.74로 가장 크게 올랐다. 특징 사이의 비선형 관계가 있는 회귀에서는 은닉층이 도움이 되었고, 이미 거의 선형 분리 가능한 분류 문제에서는 차이가 작았다. 다음 단계로 모멘텀·Adam 옵티마이저(22일차), 학습률 스케줄(23일차), 드롭아웃(26일차)을 추가한다.

    풀이
    class MomentumSGD(SGD):
    def __init__(self, params, lr=0.1, momentum=0.9, weight_decay=0.0):
        super().__init__(params, lr, weight_decay); self.m = momentum
        self.v = [np.zeros_like(p) for p, _ in params]
    def step(self):
        for (p, g), v in zip(self.params, self.v):
            v *= self.m; v += g + self.wd * p
            p -= self.lr * v

    같은 학습률에서 모멘텀이 있으면 처음 몇 에폭의 손실이 훨씬 빨리 떨어진다. 22일차에서 이유를 다룬다.

    class LeakyReLU(Module):
    def __init__(self, a=0.01): self.a = a
    def forward(self, X):
        self.mask = X > 0; return np.where(self.mask, X, self.a * X)
    def backward(self, G): return np.where(self.mask, G, self.a * G)
